local function findFirstPart(model)
    for _, descendant in ipairs(model:GetDescendants()) do
        if descendant:IsA("BasePart") then
            return descendant
        end
    end
    return nil
end

local function updateHitboxes(scale)
    for _, model in ipairs(Workspace:GetChildren()) do
        if model:IsA("Model") and model.Name:match("^CLIENT_BALL_%d+$") then
            local ball = model:FindFirstChild("Ball.001")
            if not ball then
                local basePart = findFirstPart(model)
                if basePart then
                    ball = Instance.new("Part")
                    ball.Name = "Ball.001"
                    ball.Shape = Enum.PartType.Ball
                    ball.Size = Vector3.new(2, 2, 2) * scale
                    ball.CFrame = basePart.CFrame
                    ball.Anchored = true
                    ball.CanCollide = false
                    ball.Transparency = 0.7
                    ball.Material = Enum.Material.ForceField
                    ball.Color = hitboxColor
                    ball.Parent = model
                end
            else
                ball.Size = Vector3.new(3, 3, 3) * scale
                ball.Color = hitboxColor
            end
        end
    end
end

local function removeHitboxes()
    for _, model in ipairs(Workspace:GetChildren()) do
        if model:IsA("Model") and model.Name:match("^CLIENT_BALL_%d+$") then
            local ball = model:FindFirstChild("Ball.001")
            if ball then
                ball:Destroy()
            end
        end
    end
end

local function spikeBall()
    if not silentSpikeEnabled then return end
    if spikeCooldown then return end

        findBallId()
        if not ballId then
            return
        end
    end
    if not BallService then return end

        task.wait(0.2)
        spikeCooldown = false
    end)

    local RF = BallService:FindFirstChild("RF")
    if RF then
        local Interact = RF:FindFirstChild("Interact")
        if Interact then
            Interact:InvokeServer({
                ["Charge"] = 1,
                ["Move"] = "Spike",
                ["SpecialCharge"] = 0,
                ["TiltDirection"] = Vector3.new(0, 1, 0),
                ["LookVector"] = lookVector,
                ["MoveDirection"] = Vector3.new(0, 0, 0),
                ["ClientCanRunSpecial"] = false,
                ["From"] = "Client",
                ["Timestamp"] = tick(),
                ["BallId"] = ballId,
                ["CustomClient"] = {},
            })
        end
    end
end

    spikeButton.InputBegan:Connect(function(input)

            dragging = true
            dragStart = input.Position
            startPos = spikeButton.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    
    spikeButton.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            spikeButton.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
    
    spikeButton.MouseButton1Click:Connect(function()
        spikeBall()
    end)
    
    spikeButton.Visible = false
    return spikeButton
end

local function enableRedirectSpike()
    if redirectHookActive then return end
    
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local Players = game:GetService("Players")
    local player = Players.LocalPlayer
    local camera = workspace.CurrentCamera
    
    local Knit = ReplicatedStorage.Packages._Index["sleitnick_knit@1.7.0"].knit
    local interact = Knit.Services.BallService.RF.Interact
    
    getgenv().DirectionEnabled = true
    getgenv().UseCameraDirection = true
    
    mtRedirect = getrawmetatable(game)
    oldNamecallRedirect = mtRedirect.__namecall
    setreadonly(mtRedirect, false)
    
    mtRedirect.__namecall = function(self, ...)
        local method = getnamecallmethod()
        local args = {...}
        
        if self == interact and method == "InvokeServer" then
            local data = args[1]
            if type(data) == "table" and data.Move == "Spike" and getgenv().DirectionEnabled then
                local dir
                if getgenv().UseCameraDirection and camera then
                    dir = camera.CFrame.LookVector
                else
                    dir = getgenv().SpikeDirection or Vector3.new(0, 0, -1)
                end
                data.LookVector = dir
                data.Direction = dir
                if data.MoveDirection then
                    data.MoveDirection = dir
                end
            end
            return oldNamecallRedirect(self, data)
        end
        
        return oldNamecallRedirect(self, ...)
    end
    
    setreadonly(mtRedirect, true)
    redirectHookActive = true
    redirectSpikeEnabled = true
end

local function disableRedirectSpike()
    if not redirectHookActive then return end
    
    if mtRedirect and oldNamecallRedirect then
        setreadonly(mtRedirect, false)
        mtRedirect.__namecall = oldNamecallRedirect
        setreadonly(mtRedirect, true)
    end
    
    redirectHookActive = false
    redirectSpikeEnabled = false
    getgenv().DirectionEnabled = false
    oldNamecallRedirect = nil
    mtRedirect = nil
end

                            lookVec.Z + tiltDir.Z * 0.3

                        data.TiltDirection = Vector3.new(
                            tiltDir.X * 1.35,
                            tiltDir.Y,
                            tiltDir.Z * 1.35
                        )
                    end
                end
            end
        end
        
        return oldNamecall(self, ...)
    end
    
    setreadonly(mt, true)
    tiltHookActive = true

local function getStrictFrontCourtPosition(rootPart)
    if not Court or not rootPart then return nil end
    
    local courtCFrame = Court:IsA("Model") and Court:GetPivot() or Court.CFrame
    local courtSize = Court:IsA("Model") and Court:GetExtentsSize() or Court.Size
    
    for i = 1, 20 do
        local randomX = (math.random() - 0.5) * courtSize.X
        local randomZ = (math.random() - 0.5) * courtSize.Z
        local randomOffset = Vector3.new(randomX, 0, randomZ)
        local chosenPosition = (courtCFrame * CFrame.new(randomOffset)).Position
        
        local directionToSpot = (chosenPosition - rootPart.Position).Unit
        local lookDirection = rootPart.CFrame.LookVector
        
        if lookDirection:Dot(directionToSpot) > 0.85 then
            return chosenPosition
        end
    end
    
    return rootPart.Position + rootPart.CFrame.LookVector * 10
end

local function faceStrictFrontSpotMidAir()
    local character = LocalPlayer.Character
    if not character then return end
    
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    if not rootPart then return end
    
    local targetSpot = getStrictFrontCourtPosition(rootPart)
    if not targetSpot then return end
    
    local targetLookPosition = Vector3.new(targetSpot.X, rootPart.Position.Y, targetSpot.Z)
    rootPart.CFrame = CFrame.lookAt(rootPart.Position, targetLookPosition)
end

local function dropToGround()
    local character = LocalPlayer.Character
    if not character then return end
    
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    if not rootPart then return end
    
    local rayOrigin = rootPart.Position
    local rayDirection = Vector3.new(0, -1000, 0)
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Blacklist
    raycastParams.FilterDescendantsInstances = {character}
    
    local result = workspace:Raycast(rayOrigin, rayDirection, raycastParams)
    
    if result then
        local groundPosition = result.Position + Vector3.new(0, 3, 0)
        rootPart.CFrame = CFrame.new(groundPosition) * (rootPart.CFrame - rootPart.Position)
    end
end

    local character = LocalPlayer.Character
    if not character then return end
    
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    if not rootPart then return end
    
    local rayOrigin = rootPart.Position
    local rayDirection = Vector3.new(0, -1000, 0)
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Blacklist
    raycastParams.FilterDescendantsInstances = {character}
    
    local result = workspace:Raycast(rayOrigin, rayDirection, raycastParams)
    
    if result then
        local groundPosition = result.Position + Vector3.new(0, 3, 0)
        local randomY = math.random(0, 360)
        rootPart.CFrame = CFrame.new(groundPosition) * CFrame.Angles(0, math.rad(randomY), 0)
    end
end

local function getBallJumpSpike()
    local closestBall = nil
    local closestDistance = math.huge
    
    for _, model in ipairs(workspace:GetChildren()) do
        if model:IsA("Model") and model.Name:match("^CLIENT_BALL_%d+$") then
            for _, part in ipairs(model:GetDescendants()) do
                if part:IsA("BasePart") then
                    local character = LocalPlayer.Character
                    if character then
                        local rootPart = character:FindFirstChild("HumanoidRootPart")
                        if rootPart then
                            local distance = (part.Position - rootPart.Position).Magnitude
                            if distance < closestDistance then
                                closestDistance = distance
                                closestBall = part
                            end
                        end
                    end
                    return part
                end
            end
        end
    end
    return closestBall
end

local function isBallOnMySide(ballPos)
    if not Boundary then return true end
    local boundaryPos = Boundary.Position
    local playerPos = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not playerPos then return true end
    
    local playerSide = playerPos.Position.Z > boundaryPos.Z
    local ballSide = ballPos.Z > boundaryPos.Z
    
    return playerSide == ballSide
end

local function teleportToBallJumpSpike()
    if teleportCooldown then return end
    teleportCooldown = true
    
    task.spawn(function()
        task.wait(0.5)
        teleportCooldown = false
    end)
    
    local character = LocalPlayer.Character
    if not character then 
        teleportCooldown = false
        return 
    end
    
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    if not rootPart then 
        teleportCooldown = false
        return 
    end
    
    local ball = getBallJumpSpike()
    if ball then
        if not isBallOnMySide(ball.Position) then
            teleportCooldown = false
            return
        end
        
        local currentLook = rootPart.CFrame.LookVector
        local newPosition = ball.Position - Vector3.new(0, 2, 0)
        rootPart.CFrame = CFrame.new(newPosition)
        rootPart.CFrame = CFrame.new(rootPart.Position, rootPart.Position + currentLook)
        rootPart.Velocity = Vector3.new(0, 0, 0)
        rootPart.RotVelocity = Vector3.new(0, 0, 0)
    end
    
    teleportCooldown = false
end

local function applyAirControl(rootPart)
    if bodyVelocity then return end

    bodyVelocity.Velocity = Vector3.zero

    bodyVelocity.Parent = rootPart

    if bodyVelocity then

        bodyVelocity = nil
    end
end

local function setupCharacter(character)
    local humanoid = character:WaitForChild("Humanoid")
    local rootPart = character:WaitForChild("HumanoidRootPart")
    airMovementSpeed = getWalkSpeed()

    humanoid:GetPropertyChangedSignal("Jump"):Connect(function()
        if humanoid.Jump then
            if autoShiftLock then
                task.defer(function()
                    task.wait(0.03)
                    local lookVector = Vector3.new(Camera.CFrame.LookVector.X, 0, Camera.CFrame.LookVector.Z)
                    if lookVector.Magnitude > 0 then
                        rootPart.CFrame = CFrame.lookAt(rootPart.Position, rootPart.Position + lookVector.Unit)
                        humanoid.AutoRotate = false
                    end
                end)
            else
                humanoid.AutoRotate = true
            end
        end
    end)

    humanoid.StateChanged:Connect(function(oldState, newState)
        if newState == Enum.HumanoidStateType.Freefall then
            if airMovement then
                applyAirControl(rootPart)
            end
        elseif newState == Enum.HumanoidStateType.Landed then
            removeAirControl()
            humanoid.AutoRotate = true
        end
    end)
end

local function removeLine(player)
    local data = lines[player]
    if data then
        if data.beam then data.beam:Destroy() end
        if data.target and data.target.Parent then data.target:Destroy() end
        if data.attachment and data.attachment.Parent then data.attachment:Destroy() end
        lines[player] = nil
    end
end

local function updateLine(player, index)
    if not linesEnabled then
        removeLine(player)
        return
    end

    local character = player.Character
    if not character or not character:FindFirstChild("Head") or not character:FindFirstChild("HumanoidRootPart") then
        removeLine(player)
        return
    end

    local head = character.Head
    local rootPart = character.HumanoidRootPart

    if not lines[player] then
        local attachment = Instance.new("Attachment", head)
        local target = Instance.new("Part")
        target.Anchored = true
        target.CanCollide = false
        target.Transparency = 1
        target.Size = Vector3.new(0.1, 0.1, 0.1)
        target.Parent = Workspace

        local targetAttachment = Instance.new("Attachment", target)
        local beam = Instance.new("Beam")
        beam.Attachment0 = attachment
        beam.Attachment1 = targetAttachment
        beam.Width0 = 0.25
        beam.Width1 = 0.25
        beam.FaceCamera = true
        beam.LightEmission = 1
        beam.Transparency = NumberSequence.new(0.3)
        beam.Color = ColorSequence.new(lineColors[(index - 1) % #lineColors + 1])
        beam.Parent = head

        lines[player] = { beam = beam, target = target, attachment = attachment }
    end

    local data = lines[player]
    data.target.Position = head.Position + rootPart.CFrame.LookVector * lineDistance
end

local function applyESP(player)
    if not player.Character or espHighlights[player] then return end
    local highlight = Instance.new("Highlight")
    highlight.Name = "JumpESP"
    highlight.Adornee = player.Character
    highlight.FillTransparency = 1
    highlight.OutlineTransparency = 0
    highlight.OutlineColor = Color3.fromRGB(255, 255, 0)
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Parent = player.Character
    espHighlights[player] = highlight
end

local function removeESP(player)
    if espHighlights[player] then
        espHighlights[player]:Destroy()
        espHighlights[player] = nil
    end
end

local function cleanupConnections(player)
    if espConnections[player] then
        for _, connection in pairs(espConnections[player]) do
            connection:Disconnect()
        end
        espConnections[player] = nil
    end
    removeESP(player)
end

local function setupESP(player)
    if player == LocalPlayer then return end
    local function onCharacterAdded(character)
        cleanupConnections(player)
        local humanoid = character:WaitForChild("Humanoid", 3)
        local head = character:FindFirstChild("Head")
        if not humanoid or not head then return end

        local stateConnection = humanoid.StateChanged:Connect(function(_, newState)
            if not espEnabled then return end
            if isEnemy(player) then
                if newState == Enum.HumanoidStateType.Jumping or newState == Enum.HumanoidStateType.Freefall then
                    applyESP(player)
                elseif newState == Enum.HumanoidStateType.Landed then
                    removeESP(player)
                end
            else
                removeESP(player)
            end
        end)

        local heartbeatConnection = RunService.Heartbeat:Connect(function()
            if not player.Character or not isEnemy(player) then
                removeESP(player)
            else
                local state = humanoid:GetState()
                if state ~= Enum.HumanoidStateType.Jumping and state ~= Enum.HumanoidStateType.Freefall then
                    removeESP(player)
                end
            end
        end)

        espConnections[player] = { stateConnection, heartbeatConnection }
    end

    if player.Character then
        onCharacterAdded(player.Character)
    end
    player.CharacterAdded:Connect(onCharacterAdded)
end

for _, player in ipairs(Players:GetPlayers()) do
    setupESP(player)
end

local function fetchNewCodes()
    local success, data = pcall(function()
        return game:HttpGet("https://beebom.com/haikyuu-legends-codes/")
    end)
    
    if not success or not data then
        return {}
    end
    
    local newCodes = {}
    local processed = {}
    
    for line in string.gmatch(data, "[^\n]+") do
        if string.find(line, "(NEW)") then
            local code = string.match(line, "(%u+[%u%d_]+)")
            if code and not processed[code] then
                processed[code] = true
                table.insert(newCodes, code)
            end
        end
        if #newCodes >= 3 then
            break
        end
    end
    
    return newCodes
end

            return
        end
        
        local configData = {
            Name = configName,
            HitboxScale = hitboxScale,

            LinesEnabled = linesEnabled,
            LineDistance = lineDistance,
            ESPEnabled = espEnabled,

        local success, json = pcall(function()

ConfigTab:Button({
    Title = "Load Config",
    Desc = "Load a saved config",
    Callback = function()
        if configName == "" then
            WindUI:Notify({ Title = "Error", Content = "Please enter a config name!", Duration = 3 })
            return
        end
        
        local folder = "AnonyHubConfigs"
        local filePath = folder .. "/" .. configName .. ".json"
        
        if not isfile(filePath) then
            WindUI:Notify({ Title = "Error", Content = "Config '" .. configName .. "' not found!", Duration = 3 })
            return
        end
        
        local success, data = pcall(function()
            return readfile(filePath)
        end)
        
        if success then
            local decoded = HttpService:JSONDecode(data)
            
            hitboxScale = decoded.HitboxScale or 5.0
            hitboxColorName = decoded.HitboxColor or "Green"
            local colors = {
                Green = Color3.fromRGB(0, 255, 0),
                Blue = Color3.fromRGB(0, 0, 255),
                Red = Color3.fromRGB(255, 0, 0),
                Yellow = Color3.fromRGB(255, 255, 0),
                Purple = Color3.fromRGB(128, 0, 128),
                Orange = Color3.fromRGB(255, 165, 0),
                Pink = Color3.fromRGB(255, 105, 180),
                White = Color3.fromRGB(255, 255, 255)
            }
            hitboxColor = colors[hitboxColorName] or Color3.fromRGB(0, 255, 0)
            linesEnabled = decoded.LinesEnabled ~= nil and decoded.LinesEnabled or true
            lineDistance = decoded.LineDistance or 50
            espEnabled = decoded.ESPEnabled ~= nil and decoded.ESPEnabled or true
            autoShiftLock = decoded.AutoShiftLock or false
            airMovement = decoded.AirMovement or false
            airMovementSpeed = decoded.AirMovementSpeed or 16
            silentSpikeEnabled = decoded.SilentSpike or false
            fastAttackEnabled = decoded.FastAttack or false
            fastAttackManualEnabled = decoded.FastAttackManual or false
            jumpSpikeEnabled = decoded.JumpSpike or false
            infChargeEnabled = decoded.InfCharge or false
            sanjuTiltEnabled = decoded.SanjuTilt or false
            redirectSpikeEnabled = decoded.RedirectSpike or false
            maxChargeEnabled = decoded.MaxCharge or false
            
            if sanjuTiltEnabled then
                enableSanjuTilt()
            end
            
            if redirectSpikeEnabled then
                enableRedirectSpike()
            end
            
            if maxChargeEnabled then
                enableMaxCharge()
            end
            
            if spikeButton then
                spikeButton.Visible = silentSpikeEnabled
            end
            
            if silentSpikeEnabled or fastAttackEnabled or fastAttackManualEnabled or jumpSpikeEnabled then
                enableJumps()
            end
            
            updateHitboxes(hitboxScale)
            
            WindUI:Notify({ Title = "Config Loaded", Content = "Config '" .. configName .. "' loaded!", Duration = 3 })
        else
            WindUI:Notify({ Title = "Error", Content = "Failed to load config!", Duration = 3 })
        end
    end
})

ConfigTab:Button({
    Title = "List Configs",
    Desc = "List all saved configs",
    Callback = function()
        local folder = "AnonyHubConfigs"
        if not isfolder(folder) then
            WindUI:Notify({ Title = "No Configs", Content = "No configs found!", Duration = 3 })
            return
        end
        
        local files = listfiles(folder)
        local configList = {}
        
        for _, file in ipairs(files) do
            local name = string.match(file, "([^/]+)%.json$")
            if name then
                table.insert(configList, name)
            end
        end
        
        if #configList == 0 then
            WindUI:Notify({ Title = "No Configs", Content = "No configs found!", Duration = 3 })
            return
        end
        
        local message = "Configs:\n" .. table.concat(configList, "\n")
        WindUI:Notify({ Title = "Config List", Content = message, Duration = 5 })
    end
})
